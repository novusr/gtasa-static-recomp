cmake_minimum_required(VERSION 3.21)
project(RemillInstalledConsumer C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(remill CONFIG REQUIRED)
add_executable(remill-installed-consumer probe.cpp)
target_link_libraries(remill-installed-consumer PRIVATE remill)
target_include_directories(remill-installed-consumer SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
